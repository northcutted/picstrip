import UIKit
import XCTest

/// Fresh English screenshots for this review, independent of Fastlane's saved locale.
@MainActor
final class ReleaseReviewScreenshots: XCTestCase {
    private func capture(_ name: String) {
        let attachment = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
        attachment.name = name
        attachment.lifetime = .keepAlways
        add(attachment)
    }

    func testCurrentFiveStepFlow() throws {
        continueAfterFailure = false
        let app = XCUIApplication()
        app.launchArguments = ["-AppleLanguages", "(en)", "-AppleLocale", "en_US"]
        app.launchEnvironment["PICSTRIP_FORCE_SCAN_BUTTON"] = "1"
        app.launchEnvironment["PICSTRIP_DISABLE_NAME_DETECTION"] = "1"
        app.launch()
        XCTAssertTrue(app.buttons["selectPhotoButton"].waitForExistence(timeout: 15))
        capture("01-Home")
        app.buttons["infoButton"].tap()
        capture("02-About")
        app.terminate()

        let url = try XCTUnwrap(Bundle(for: Self.self).url(forResource: "test_list", withExtension: "png"))
        let path = "/tmp/picstrip_review_fixture.png"
        try Data(contentsOf: url).write(to: URL(fileURLWithPath: path))
        app.launchEnvironment["PICSTRIP_FIXTURE"] = path
        app.launch()
        let edit = app.descendants(matching: .any)["editRedactionsButton"]
        XCTAssertTrue(edit.waitForExistence(timeout: 25))
        capture("03-PhotoLoaded")
        edit.tap()
        let add = app.descendants(matching: .any)["addRedactionButton"]
        XCTAssertTrue(add.waitForExistence(timeout: 5))
        Thread.sleep(forTimeInterval: 2)
        capture("04a-RedactionEditorIdle")
        add.tap()
        let preview = app.descendants(matching: .any)["metadataPhotoPreview"]
        preview.coordinate(withNormalizedOffset: CGVector(dx: 0.30, dy: 0.30))
            .press(forDuration: 0.1, thenDragTo: preview.coordinate(withNormalizedOffset: CGVector(dx: 0.58, dy: 0.45)))
        Thread.sleep(forTimeInterval: 2)
        capture("04-RedactionEditor")
        app.descendants(matching: .any)["doneEditingRedactionsButton"].tap()
        app.buttons["saveButton"].tap()
        XCTAssertTrue(app.descendants(matching: .any)["savePreviewImage"].waitForExistence(timeout: 10))
        capture("05-ReviewAndSave")
    }
}
